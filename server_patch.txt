CHITAI_BASE = "https://www.chitai-gorod.ru"

def chitai_fetch(url, timeout=7):
    req = urllib.request.Request(url, headers={
        "User-Agent": "Mozilla/5.0 (Linux; Android 15) AppleWebKit/537.36 Chrome/130 Mobile Safari/537.36",
        "Accept": "text/html,application/xhtml+xml,application/json;q=0.9,*/*;q=0.8",
        "Accept-Language": "ru-RU,ru;q=0.9,en;q=0.7",
        "Referer": CHITAI_BASE + "/",
    })
    with urllib.request.urlopen(req, timeout=timeout) as r:
        return r.status, r.geturl(), r.read().decode("utf-8", "ignore")

def chitai_product(url):
    try:
        _, final_url, page = chitai_fetch(url)
    except Exception:
        return None

    title = author = isbn = cover = publisher = ""
    images = []

    # JSON-LD
    for raw in re.findall(
        r'<script\b[^>]*type=["\']application/ld\+json["\'][^>]*>(.*?)</script>',
        page, re.I | re.S
    ):
        try:
            obj = json.loads(raw.strip())
        except Exception:
            continue
        objs = obj if isinstance(obj, list) else [obj]
        for o in objs:
            if not isinstance(o, dict):
                continue
            if not title and o.get("name"):
                title = str(o["name"]).strip()

            a = o.get("author")
            if not author:
                if isinstance(a, dict):
                    author = str(a.get("name", "")).strip()
                elif isinstance(a, list):
                    author = ", ".join(
                        str(v.get("name", "")) for v in a
                        if isinstance(v, dict)
                    ).strip(", ")
                elif isinstance(a, str):
                    author = a.strip()

            im = o.get("image")
            if im:
                if isinstance(im, list):
                    images.extend(str(v) for v in im)
                else:
                    images.append(str(im))

    # OpenGraph
    def meta(name):
        m = re.search(
            r'<meta\b[^>]*(?:property|name)=["\']' + re.escape(name) +
            r'["\'][^>]*content=["\']([^"\']+)',
            page, re.I
        )
        return html.unescape(m.group(1)).strip() if m else ""

    if not title:
        title = meta("og:title")
    if not cover:
        cover = meta("og:image")

    # ISBN
    m = re.search(
        r'ISBN[^0-9Xx]{0,60}([0-9Xx][0-9Xx\- ]{9,20})',
        page, re.I
    )
    if m:
        isbn = clean_isbn(m.group(1))

    if not author:
        m = re.search(
            r'(?:Автор|Авторы)\s*</?[^>]*>\s*([^<]{2,160})',
            page, re.I | re.S
        )
        if m:
            author = megaknigi_clean(m.group(1))

    # Собираем URL изображений из страницы.
    for m in re.finditer(
        r'https?://[^"\']+\.(?:jpg|jpeg|png|webp)(?:\?[^"\']*)?',
        page, re.I
    ):
        images.append(html.unescape(m.group(0)))

    if not cover and images:
        cover = images[0]

    if not title:
        return None

    gallery = []
    for u in [cover] + images:
        u = urllib.parse.urljoin(CHITAI_BASE, u)
        if u and u not in gallery:
            gallery.append(u)

    return {
        "title": title,
        "author": author,
        "isbn": isbn,
        "cover": cover,
        "gallery": gallery[:12],
        "publisher": publisher,
        "source": "Читай-город",

def chitaigorod(q, limit=10):
    q = (q or "").strip()
    if not q:
        return []

    isbn = clean_isbn(q)
    url = CHITAI_BASE + "/search?" + urllib.parse.urlencode({
        "q": isbn or q
    })

    try:
        _, _, page = chitai_fetch(url)
    except Exception:
        return []

    paths = []
    for m in re.finditer(
        r'href=["\'](/product/[^"\']+)',
        page, re.I
    ):
        p = m.group(1).split("?")[0]
        if p not in paths:
            paths.append(p)
        if len(paths) >= 15:
            break

    out = []
    seen = set()

    for p in paths:
        item = chitai_product(urllib.parse.urljoin(CHITAI_BASE, p))
        if not item:
            continue

        # Для ISBN — только ТОЧНОЕ совпадение.
        if isbn and clean_isbn(item.get("isbn", "")) != isbn:
            continue

        key = (
            clean_isbn(item.get("isbn", "")),
            norm(item.get("title", "")),
            norm(item.get("author", ""))
        )
        if key in seen:
            continue

        seen.add(key)
        out.append(item)

        if len(out) >= limit:
            break

    return out
